function MapScreen({ mode, language, t, onBack }: {
  mode: MapMode
  language: AppLanguage
  t: MobileCopy
  onBack: () => void
}) {
  const { query, mapPolygon, setMapPolygon, clearMapPolygon, filters, setFilters, saveCurrentSearch } = useApp()
  const mapRef = useRef<google.maps.Map | null>(null)
  const fittedPolygonRef = useRef('')
  const [mapType, setMapType] = useState<'roadmap' | 'hybrid'>('roadmap')
  const [saved, setSaved] = useState(false)
  const [mapStatus, setMapStatus] = useState<MapStatus>('loading')
  const [locationStatus, setLocationStatus] = useState<LocationStatus>('idle')
  const [locationOverride, setLocationOverride] = useState('')
  const [drawing, setDrawing] = useState(false)
  const [boundsDirty, setBoundsDirty] = useState(false)
  const [searchBounds, setSearchBounds] = useState<MobileMapBounds | null>(null)
  const [visibleCount, setVisibleCount] = useState(0)
  const polygonSignature = mapPolygon.map((point) => `${point.lat.toFixed(5)},${point.lng.toFixed(5)}`).join(';')

  const toggleLayers = () => {
    const next = mapType === 'roadmap' ? 'hybrid' : 'roadmap'
    setMapType(next)
    mapRef.current?.setMapTypeId(next)
  }

  useEffect(() => {
    const map = mapRef.current
    if (!map || mapStatus !== 'ready' || drawing) return
    const dirty = () => setBoundsDirty(true)
    const dragListener = map.addListener('dragstart', dirty)
    const div = map.getDiv()
    div.addEventListener('wheel', dirty, { passive: true })
    div.addEventListener('dblclick', dirty)
    return () => {
      dragListener.remove()
      div.removeEventListener('wheel', dirty)
      div.removeEventListener('dblclick', dirty)
    }
  }, [drawing, mapStatus])

  useEffect(() => {
    setSearchBounds(null)
    setBoundsDirty(false)
    setSaved(false)
  }, [polygonSignature, query])

  useEffect(() => {
    const map = mapRef.current
    if (!map || mapStatus !== 'ready' || mapPolygon.length < 3 || fittedPolygonRef.current === polygonSignature) return
    fittedPolygonRef.current = polygonSignature
    const bounds = new google.maps.LatLngBounds()
    mapPolygon.forEach((point) => bounds.extend(point))
    map.fitBounds(bounds, { top: 88, right: 28, bottom: 128, left: 28 })
  }, [mapPolygon, mapStatus, polygonSignature])

  const locate = () => {
    if (!navigator.geolocation) {
      setLocationOverride('')
      setLocationStatus('error')
      return
    }
    setLocationOverride('')
    setLocationStatus('loading')
    navigator.geolocation.getCurrentPosition(({ coords }) => {
      const current = { lat: coords.latitude, lng: coords.longitude }
      if (!isInsideTenerife(current)) {
        setLocationOverride(t.outsideTenerife)
        setLocationStatus('error')
        return
      }
      mapRef.current?.panTo(current)
      mapRef.current?.setZoom(14)
      setBoundsDirty(true)
      setLocationStatus('success')
    }, () => {
      setLocationOverride('')
      setLocationStatus('error')
    }, { enableHighAccuracy: true, timeout: 10_000, maximumAge: 60_000 })
  }

  const locationMessage = locationOverride || (locationStatus === 'loading' ? t.locating : locationStatus === 'success' ? t.locationFound : locationStatus === 'error' ? t.locationDenied : '')
  const mainDrawLabel = drawing ? t.cancelDrawing : mapPolygon.length >= 3 ? t.redrawZone : t.drawZone

  const applyPolygon = (points: MapPoint[]) => {
    setSearchBounds(null)
    setBoundsDirty(false)
    if (points.length >= 3) {
      if (filters.areas.length) setFilters({ ...filters, areas: [] })
      setMapPolygon(points)
    } else {
      clearMapPolygon()
    }
  }

  const toggleDrawing = () => {
    setSearchBounds(null)
    setBoundsDirty(false)
    setDrawing((current) => !current)
  }

  const searchVisibleArea = () => {
    const current = mapRef.current?.getBounds()
    if (!current) return
    const northEast = current.getNorthEast()
    const southWest = current.getSouthWest()
    setSearchBounds({ north: northEast.lat(), east: northEast.lng(), south: southWest.lat(), west: southWest.lng() })
    setBoundsDirty(false)
  }

  const openResults = (panel: 'results' | 'filters') => {
    window.dispatchEvent(new CustomEvent('112233:open-mobile-results', { detail: { panel } }))
  }

  const saveSearch = () => {
    saveCurrentSearch()
    setSaved(true)
  }

  return <section className={cn('m2-map-screen', drawing && 'is-freehand-drawing', mapPolygon.length >= 3 && 'has-drawn-zone')} data-testid={`map-${mode}`}>
    {mode === 'draw' ? <BackHeader title={t.mapDrawTitle} onBack={onBack} backLabel={t.back} /> : <><header className="m2-map-results-header"><button type="button" className="m2-icon-button" onClick={onBack} aria-label={t.back}><ChevronLeft /></button><div><strong>Tenerife</strong><small>{query || t.visibleArea}</small></div><button type="button" className={cn('m2-save', saved && 'is-saved')} onClick={saveSearch} aria-pressed={saved}>{saved ? <Check /> : <Bell />}{saved ? t.saved : t.save}</button></header><div className="m2-map-toolbar"><button type="button" onClick={() => openResults('filters')}><SlidersHorizontal />{t.filters}</button><button type="button" onClick={() => openResults('results')}><Menu />{t.list}</button><span className="m2-map-toolbar__count" data-testid="mobile-map-count">{visibleCount}</span></div></>}
    <GoogleMapCanvas language={language} t={t} mapRef={mapRef} query={query} onStatus={setMapStatus} />
    <MobileMapListingsLayer mapRef={mapRef} mapReady={mapStatus === 'ready'} language={language} drawing={drawing} query={query} polygon={mapPolygon} bounds={searchBounds} onVisibleCount={setVisibleCount} />
    <FreehandAreaLayer mapRef={mapRef} mapReady={mapStatus === 'ready'} active={drawing} setActive={setDrawing} polygon={mapPolygon} onPolygonChange={applyPolygon} t={t} />
    {boundsDirty && !drawing ? <button type="button" className="m2-map-search-area" data-testid="mobile-map-search-area" onClick={searchVisibleArea}><Search />{t.searchThisArea}</button> : null}
    <div className="m2-map-controls"><button type="button" onClick={toggleLayers} aria-label={t.layers} aria-pressed={mapType === 'hybrid'} disabled={mapStatus !== 'ready' || drawing}><Layers3 /></button><button type="button" onClick={locate} aria-label={t.locate} disabled={mapStatus !== 'ready' || locationStatus === 'loading' || drawing}><Crosshair /></button></div>
    {locationMessage ? <div className={cn('m2-location-toast', locationStatus === 'error' && 'is-error')} role="status">{locationMessage}</div> : null}
    <div className="m2-draw-actions">{mapPolygon.length >= 3 && !drawing ? <button type="button" className="m2-clear-zone" onClick={() => applyPolygon([])} aria-label={t.clearZone}><Trash2 /></button> : null}<button type="button" className="m2-draw-cta" onClick={toggleDrawing} disabled={mapStatus !== 'ready'} aria-pressed={drawing}><PenTool />{mainDrawLabel}</button></div>
  </section>
}
