  useEffect(() => {
    const openResults = (event: Event) => {
      const detail = (event as CustomEvent<{ panel?: ResultsPanel; listingId?: string }>).detail
      setFilters((current) => ({ ...current, rentalMode }))
      setFocusListingId(detail?.listingId ?? '')
      setLanguage(currentLanguage())
      setPanel(detail?.panel === 'filters' || detail?.panel === 'sort' ? detail.panel : 'results')
      setOpen(true)
    }
    window.addEventListener('112233:open-mobile-results', openResults)
    return () => window.removeEventListener('112233:open-mobile-results', openResults)
  }, [rentalMode])
